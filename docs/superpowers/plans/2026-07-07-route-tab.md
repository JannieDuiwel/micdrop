# Route Tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A "Route" tab that captures live audio playing on a chosen playback device (WASAPI loopback) and mixes it into MicDrop's Mic output so voice-party friends hear the music.

**Architecture:** A `RingBuffer` + pure `mix_block()` function in `audio.py` let the existing per-device output callback mix a live feed with clip playback. A new `routing.Router` runs a capture thread using the `soundcard` library (loopback) and pushes gained blocks into the Player's primary voice. The Tkinter window gains a `ttk.Notebook` with Soundboard and Route tabs.

**Tech Stack:** Python 3.13, Tkinter/ttk, sounddevice (output, unchanged), soundcard 0.4.6 (loopback capture), numpy, pytest.

## Global Constraints

- **No git repository exists** in this project — skip all commit steps (if the user runs `git init` later, commit after each task instead).
- New dependency: `soundcard>=0.4.6` — no other new runtime dependencies.
- Spec: `docs/superpowers/specs/2026-07-07-route-tab-design.md`.
- Python/venv: run everything with `.venv\Scripts\python.exe` from the project root.
- Capture opens at the mic voice's samplerate (WASAPI shared mode converts); never resample routed audio in our code.
- Routing always starts OFF at app launch; config only remembers source + volume.
- Verified on this machine: `soundcard.all_speakers()` lists playback devices; `sc.get_microphone(spk.id, include_loopback=True).recorder(samplerate=48000, channels=2, blocksize=1024)` works from a worker thread and `record(n)` returns float32 `(n, 2)`.
- Known limitation (accept, don't fight): WASAPI loopback delivers no packets while the source is silent, so `record()` can block; `Router.stop()` joins with a timeout and abandons the (daemon) thread — a generation token makes stale threads inert.

```python
class RingBuffer:
    """Fixed-capacity float32 frame buffer; one writer thread, one reader thread.

    `write` drops the oldest frames on overflow; `read` returns fewer frames
    than asked on underrun (possibly zero rows).
    """

    def __init__(self, capacity_frames: int, channels: int) -> None:
        self._buf = np.zeros((capacity_frames, channels), dtype="float32")
        self._cap = capacity_frames
        self.channels = channels
        self._r = 0  # absolute frames read
        self._w = 0  # absolute frames written
        self._lock = threading.Lock()

    def write(self, data: np.ndarray) -> None:
        with self._lock:
            n = data.shape[0]
            if n >= self._cap:
                data = data[-self._cap :]
                n = data.shape[0]
            start = self._w % self._cap
            first = min(n, self._cap - start)
            self._buf[start : start + first] = data[:first]
            if n > first:
                self._buf[: n - first] = data[first:]
            self._w += n
            if self._w - self._r > self._cap:
                self._r = self._w - self._cap

    def read(self, frames: int) -> np.ndarray:
        with self._lock:
            n = min(frames, self._w - self._r)
            out = np.empty((n, self.channels), dtype="float32")
            start = self._r % self._cap
            first = min(n, self._cap - start)
            out[:first] = self._buf[start : start + first]
            if n > first:
                out[first:] = self._buf[: n - first]
            self._r += n
            return out


def mix_block(
    outdata: np.ndarray,
    buf: np.ndarray | None,
    pos: int,
    live: RingBuffer | None,
) -> tuple[np.ndarray | None, int]:
    """Fill one output block from the clip buffer plus the live feed.

    Returns the advanced (buf, pos); buf becomes None when the clip ends.
    Whenever live audio is mixed in, the result is hard-clipped to ±1.
    """
    frames = outdata.shape[0]
    if buf is None:
        outdata.fill(0)
    else:
        chunk = buf[pos : pos + frames]
        n = chunk.shape[0]
        if n:
            outdata[:n] = chunk
        if n < frames:
            outdata[n:] = 0
            buf, pos = None, 0
        else:
            pos += frames
    if live is not None:
        lchunk = live.read(frames)
        m = lchunk.shape[0]
        if m:
            outdata[:m] += lchunk
            np.clip(outdata, -1.0, 1.0, out=outdata)
    return buf, pos
```

```python
    def enable_live(self) -> tuple[int, int]:
        """Attach a live ring buffer (~1 s) to the mic voice for routed audio.

        Returns (samplerate, channels) the feed must match.
        """
        with self._lock:
            v = self._primary
            if v is None:
                raise RuntimeError("No output device selected")
            v.live = RingBuffer(v.samplerate, v.channels)
            return v.samplerate, v.channels

    def feed_live(self, data: np.ndarray) -> None:
        v = self._primary
        if v is not None and v.live is not None:
            v.live.write(data)

    def disable_live(self) -> None:
        with self._lock:
            if self._primary is not None:
                self._primary.live = None
```

```python
"""Live loopback routing.

Captures what is playing on a chosen playback device (WASAPI loopback via the
`soundcard` library) on a worker thread and feeds gained float32 blocks into
the Player's mic voice, where they are mixed with clip playback.

soundcard is imported lazily so the app still starts if it is missing/broken;
the Route tab then reports the error instead of crashing on launch.
"""

from __future__ import annotations

import contextlib
import threading
import warnings
from collections.abc import Callable

import numpy as np

BLOCK_FRAMES = 960  # ~20 ms at 48 kHz: low latency without hammering the CPU


def list_route_sources() -> list[str]:
    """Names of playback devices whose audio can be captured via loopback."""
    import soundcard as sc

    return [s.name for s in sc.all_speakers()]


@contextlib.contextmanager
def _open_loopback_recorder(source_name: str, samplerate: int, blocksize: int):
    import soundcard as sc

    warnings.filterwarnings("ignore", module="soundcard")  # capture-gap noise
    spk = next((s for s in sc.all_speakers() if s.name == source_name), None)
    if spk is None:
        raise RuntimeError(f"Playback device not found: {source_name}")
    mic = sc.get_microphone(spk.id, include_loopback=True)
    with mic.recorder(samplerate=samplerate, channels=2, blocksize=blocksize) as rec:
        yield rec


class Router:
    """Owns one capture thread routing a playback device into the mic feed.

    WASAPI loopback yields no packets while the source is silent, so a
    `record()` call can block indefinitely; `stop()` therefore joins with a
    timeout and abandons the daemon thread. A generation token keeps such a
    stale thread from feeding audio or raising errors afterwards.
    """

    def __init__(
        self,
        feed: Callable[[np.ndarray], None],
        on_error: Callable[[str], None],
        open_recorder=None,
    ) -> None:
        self._feed = feed
        self._on_error = on_error
        self._open_recorder = open_recorder or _open_loopback_recorder
        self._thread: threading.Thread | None = None
        self._gen = 0
        self._gain = 1.0
        self.peak = 0.0
        self.source_name = ""

    @property
    def running(self) -> bool:
        t = self._thread
        return t is not None and t.is_alive()

    def set_gain(self, gain: float) -> None:
        self._gain = max(0.0, min(2.0, float(gain)))

    def start(self, source_name: str, samplerate: int, channels: int) -> None:
        self.stop()
        self._gen += 1
        self.source_name = source_name
        self._thread = threading.Thread(
            target=self._run,
            args=(self._gen, source_name, samplerate, channels),
            daemon=True,
        )
        self._thread.start()

    def stop(self) -> None:
        self._gen += 1  # invalidate the running thread immediately
        t = self._thread
        if t is not None:
            t.join(timeout=1.0)
        self._thread = None
        self.peak = 0.0
        self.source_name = ""

    def _run(self, gen: int, source_name: str, samplerate: int, channels: int) -> None:
        try:
            with self._open_recorder(source_name, samplerate, BLOCK_FRAMES) as rec:
                while gen == self._gen:
                    data = rec.record(numframes=BLOCK_FRAMES)
                    if gen != self._gen:
                        break
                    block = np.asarray(data, dtype="float32") * self._gain
                    if channels == 1 and block.ndim == 2 and block.shape[1] > 1:
                        block = block.mean(axis=1, keepdims=True)
                    self.peak = float(np.max(np.abs(block))) if block.size else 0.0
                    self._feed(block)
        except Exception as exc:  # noqa: BLE001 - any capture failure ends routing
            if gen == self._gen:
                self.peak = 0.0
                self._on_error(f"Routing stopped: {exc}")
```

```python
        self.route_combo.grid(row=1, column=1, sticky="we", padx=(6, 0))
        self.route_combo.bind("<<ComboboxSelected>>", self._on_route_source_select)

        ttk.Label(f, text="Music volume:").grid(row=2, column=0, sticky="w", pady=(10, 0))
        self.route_vol_scale = ttk.Scale(
            f, from_=0, to=100, orient="horizontal", command=self._on_route_volume
        )
        self.route_vol_scale.set(self.cfg.route_volume * 100)
        self.route_vol_scale.grid(row=2, column=1, sticky="we", padx=(6, 0), pady=(10, 0))
        self.route_vol_scale.bind("<ButtonRelease-1>", lambda e: self._save())

        self.route_btn = ttk.Button(
            f, text="▶  Route to mic", style="Accent.TButton", command=self._toggle_route
        )
        self.route_btn.grid(row=3, column=0, columnspan=2, sticky="we", pady=(14, 0))

        ttk.Label(f, text="Level:").grid(row=4, column=0, sticky="w", pady=(10, 0))
        self.route_meter = tk.Canvas(
            f, height=12, bg=self.palette["trough"], highlightthickness=0
        )
        self.route_meter.grid(row=4, column=1, sticky="we", padx=(6, 0), pady=(10, 0))

        self.route_status_var = tk.StringVar(value="Not routing.")
        ttk.Label(f, textvariable=self.route_status_var, style="Muted.TLabel").grid(
            row=5, column=0, columnspan=2, sticky="w", pady=(10, 0)
        )
        f.columnconfigure(1, weight=1)

        self._load_route_sources()

    def _load_route_sources(self) -> None:
        """Fill the source dropdown, excluding the current mic output (feedback loop)."""
        try:
            names = routing.list_route_sources()
        except Exception as exc:  # noqa: BLE001 - soundcard missing/broken
            names = []
            self._set_status(f"Can't list routable devices: {exc}", 6)
        # Exclude the *applied* mic device (auto-detected ones never reach cfg).
        mic = ""
        for d in self._devices:
            if d.index == self.player.device_index:
                mic = d.name.strip().lower()
                break
        if mic:
            names = [
                n
                for n in names
                if not (n.strip().lower().startswith(mic) or mic.startswith(n.strip().lower()))
            ]
        self._route_sources = names
        if names:
            self.route_combo["values"] = names
            self.route_btn.state(["!disabled"])
            saved = self.cfg.route_source_name
            self.route_source_var.set(saved if saved in names else names[0])
        else:
            self.route_combo["values"] = [self.ROUTE_PLACEHOLDER]
            self.route_source_var.set(self.ROUTE_PLACEHOLDER)
            self.route_btn.state(["disabled"])

    def _on_route_source_select(self, _event: tk.Event) -> None:
        name = self.route_source_var.get()
        if name == self.ROUTE_PLACEHOLDER:
            return
        self.cfg.route_source_name = name
        self._save()
        if self.router.running:
            self._stop_route()
            self._start_route()
        self._update_route_hint()

    def _update_route_hint(self) -> None:
        src = self.route_source_var.get().strip().lower()
        mon = self.cfg.monitor_device_name.strip().lower()
        if mon and src and (src.startswith(mon) or mon.startswith(src)):
            self.route_status_var.set(
                "Note: this is also your monitor device — friends will hear clips twice (slight echo)."
            )
        elif not self.router.running:
            self.route_status_var.set("Not routing.")

    def _start_route(self) -> None:
        name = self.route_source_var.get()
        if not name or name == self.ROUTE_PLACEHOLDER:
            self._set_status("Pick a device to route first", 4)
            return
        try:
            sr, ch = self.player.enable_live()
        except RuntimeError:
            self._set_status("Select a Mic output device first", 4)
            return
        self.router.set_gain(self.cfg.route_volume)
        self.router.start(name, sr, ch)
        self.cfg.route_source_name = name
        self._save()
        self.route_btn.configure(text="■  Stop routing", style="Danger.TButton")
        mic = self.cfg.output_device_name or "mic output"
        self.route_status_var.set(f"Routing: {name}  →  {mic}")

    def _stop_route(self) -> None:
        self.router.stop()
        self.player.disable_live()
        self.route_btn.configure(text="▶  Route to mic", style="Accent.TButton")
        self.route_status_var.set("Not routing.")
        self._draw_route_meter(0.0)

    def _toggle_route(self) -> None:
        if self.router.running:
            self._stop_route()
        else:
            self._start_route()

    def _on_route_volume(self, value: str) -> None:
        v = float(value) / 100.0
        self.cfg.route_volume = v
        self.router.set_gain(v)

    def _draw_route_meter(self, level: float) -> None:
        c = self.route_meter
        c.delete("all")
        w = c.winfo_width()
        if w > 1 and level > 0:
            c.create_rectangle(
                0, 0, int(w * min(1.0, level)), 12, fill=self.palette["accent"], width=0
            )
```
